Лабораторная работа №1 - Абстрактный алфавит

Задание

Написать программу, которая кодирует и декодирует текст тремя способами: латинский текст - кодом ASCII и кодом Trisime, русский текст - кодом Морзе. Таблицы соответствия (код Морзе для русского алфавита, код Trisime для латиницы, расширенный ASCII) даны в методичке.

Теоретическая часть

Кодирование - это замена символов одного алфавита на комбинации символов другого алфавита по фиксированному правилу (таблице соответствия).

ASCII: каждой латинской букве присвоено число (например H = 72), которое печатается в двоичном виде по 8 бит на символ. Декодирование - обратная операция: каждые 8 бит переводятся в число, по числу находится символ. Код равномерный - длина комбинации всегда 8 бит, независимо от буквы.

Trisime: тоже равномерный код, но алфавит комбинаций - не 0/1, а три цифры 1, 2, 3, длина комбинации всегда 3 символа. 3 позиции по 3 варианта дают 3^3 = 27 комбинаций, а букв в латинице 26 плюс точка - как раз хватает. Кодирование - поиск буквы в таблице, декодирование - поиск комбинации в обратной таблице.

Код Морзе (русский алфавит): неравномерный код - частым буквам (Е, И, Т) даны короткие комбинации, редким (Ш, Щ, Ъ) - длинные, чтобы среднее сообщение передавалось быстрее. Кодирование - та же замена по таблице; при декодировании между буквами нужен разделитель (пробел), иначе комбинацию нельзя однозначно разбить на буквы.

Листинг программы (C++, среда - clang++, Delphi 7 недоступен)

```cpp
#include <iostream>
#include <string>
#include <vector>
#include <map>
#include <sstream>
#include <cctype>

using namespace std;

vector<char32_t> utf8_decode(const string& s) {
    vector<char32_t> out;
    size_t i = 0;
    while (i < s.size()) {
        unsigned char c = (unsigned char)s[i];
        char32_t cp; int len;
        if (c < 0x80) { cp = c; len = 1; }
        else if ((c >> 5) == 0x6) { cp = c & 0x1F; len = 2; }
        else if ((c >> 4) == 0xE) { cp = c & 0x0F; len = 3; }
        else { cp = c & 0x07; len = 4; }
        for (int k = 1; k < len && i + k < s.size(); k++)
            cp = (cp << 6) | (((unsigned char)s[i + k]) & 0x3F);
        out.push_back(cp);
        i += len;
    }
    return out;
}

string utf8_encode(char32_t cp) {
    string out;
    if (cp < 0x80) {
        out += (char)cp;
    } else if (cp < 0x800) {
        out += (char)(0xC0 | (cp >> 6));
        out += (char)(0x80 | (cp & 0x3F));
    } else {
        out += (char)(0xE0 | (cp >> 12));
        out += (char)(0x80 | ((cp >> 6) & 0x3F));
        out += (char)(0x80 | (cp & 0x3F));
    }
    return out;
}

string ascii_encode(const string& text) {
    ostringstream out;
    for (unsigned char c : text) {
        for (int bit = 7; bit >= 0; bit--) out << ((c >> bit) & 1);
        out << ' ';
    }
    return out.str();
}

string ascii_decode(const string& code) {
    istringstream in(code);
    string group, out;
    while (in >> group) {
        unsigned char c = 0;
        for (char b : group) c = (c << 1) | (b - '0');
        out += (char)c;
    }
    return out;
}

map<char, string> trisimeTable = {
    {'A',"111"},{'B',"112"},{'C',"113"},{'D',"121"},{'E',"122"},{'F',"123"},
    {'G',"131"},{'H',"132"},{'I',"133"},{'J',"211"},{'K',"212"},{'L',"213"},
    {'M',"221"},{'N',"222"},{'O',"223"},{'P',"231"},{'Q',"232"},{'R',"233"},
    {'S',"311"},{'T',"312"},{'U',"313"},{'V',"321"},{'W',"322"},{'X',"323"},
    {'Y',"331"},{'Z',"332"},{'.',"333"}
};
map<string, char> trisimeReverse;
void buildTrisimeReverse() {
    for (auto& p : trisimeTable) trisimeReverse[p.second] = p.first;
}

string trisime_encode(const string& text) {
    ostringstream out;
    for (char c : text) {
        char up = toupper((unsigned char)c);
        auto it = trisimeTable.find(up);
        if (it != trisimeTable.end()) out << it->second << ' ';
    }
    return out.str();
}

string trisime_decode(const string& code) {
    istringstream in(code);
    string group, out;
    while (in >> group) {
        if (group == "000") { out += ' '; continue; }
        auto it = trisimeReverse.find(group);
        if (it != trisimeReverse.end()) out += it->second;
    }
    return out;
}

map<char32_t, string> morseTable = {
    {U'А',".-"},   {U'Б',"-..."}, {U'В',".--"},  {U'Г',"--."},  {U'Д',"-.."},
    {U'Е',"."},    {U'Ж',"...-"}, {U'З',"--.."}, {U'И',".."},   {U'Й',".---"},
    {U'К',"-.-"},  {U'Л',".-.."}, {U'М',"--"},   {U'Н',"-."},   {U'О',"---"},
    {U'П',".--."}, {U'Р',".-."},  {U'С',"..."},  {U'Т',"-"},    {U'У',"..-"},
    {U'Ф',"..-."}, {U'Х',"...."}, {U'Ц',"-.-."}, {U'Ч',"---."}, {U'Ш',"----"},
    {U'Щ',"--.-"}, {U'Ъ',"--.--"},{U'Ы',"-.--"}, {U'Ь',"-..-"}, {U'Э',"..-.."},
    {U'Ю',"..--"}, {U'Я',".-.-"}
};
map<string, char32_t> morseReverse;
void buildMorseReverse() {
    for (auto& p : morseTable) morseReverse[p.second] = p.first;
}

string morse_encode(const string& text_utf8) {
    vector<char32_t> chars = utf8_decode(text_utf8);
    ostringstream out;
    for (char32_t c : chars) {
        if (c == U' ') { out << "/ "; continue; } 
        auto it = morseTable.find(c);
        if (it != morseTable.end()) out << it->second << ' ';
    }
    return out.str();
}

string morse_decode(const string& code) {
    istringstream in(code);
    string group, out;
    while (in >> group) {
        if (group == "/") { out += ' '; continue; }
        auto it = morseReverse.find(group);
        if (it != morseReverse.end()) out += utf8_encode(it->second);
    }
    return out;
}

int main() {
    buildTrisimeReverse();
    buildMorseReverse();

    cout << "=== 1. ASCII код (латиница) ===\n";
    string latText = "HELLO";
    string ac = ascii_encode(latText);
    cout << "Исходный текст: " << latText << "\n";
    cout << "Код (8 бит/симв.): " << ac << "\n";
    cout << "Декодировано: " << ascii_decode(ac) << "\n\n";

    cout << "=== 2. Код Trisime (латиница) ===\n";
    string latText2 = "CAB";
    string tc = trisime_encode(latText2);
    cout << "Исходный текст: " << latText2 << "\n";
    cout << "Код (3 симв./букву): " << tc << "\n";
    cout << "Декодировано: " << trisime_decode(tc) << "\n\n";

    cout << "=== 3. Код Морзе (русский алфавит) ===\n";
    string ruText = "МАМА МЫЛА РАМУ";
    string mc = morse_encode(ruText);
    cout << "Исходный текст: " << ruText << "\n";
    cout << "Код: " << mc << "\n";
    cout << "Декодировано: " << morse_decode(mc) << "\n";

    return 0;
}
```

Ответы на контрольные вопросы

1) Как определяется алфавит?
Алфавит - конечный набор различимых символов (букв), из которых по определённым правилам составляются сообщения. Сам по себе алфавит ничего не значит - соответствие символам одного алфавита символов другого задаёт код. Например, латинский алфавит - это просто 26 букв A-Z, а то, что H кодируется как 01001000 - уже свойство конкретного кода (ASCII), а не алфавита самого по себе.

2) Что такое равномерный и неравномерный код?
Равномерный код - все кодовые комбинации имеют одинаковую длину. Пример - Trisime: у любой буквы код ровно 3 цифры (111, 231, 332 - всегда 3 знака). ASCII тоже равномерный: всегда 8 бит.
Неравномерный код - длина комбинаций разная для разных букв. Пример - код Морзе: буква Е - это одна точка, буква Ш - четыре тире. Неравномерность вводится специально: часто встречающимся буквам дают короткие коды, чтобы среднее сообщение получалось короче - та же идея в более развитой форме используется в кодах Хаффмана и Шеннона-Фано.

3) Как можно объяснить высокую помехоустойчивость сообщений, передаваемых на русском языке?
Дело в избыточности естественного языка. У реального текста очень неравномерное распределение букв и сильные связи между соседними буквами (после "Ъ" почти никогда не идёт "Ъ" снова, сочетания вроде "тся"/"ться" предсказуемы и т.д.). Из-за этого реальная энтропия русского текста намного меньше максимально возможной (log2 от размера алфавита) - избыточность русского языка оценивается примерно в 70-80%. Поэтому если часть сообщения потерялась или исказилась из-за помех, оставшуюся часть часто можно однозначно восстановить по контексту и грамматике - язык сам несёт встроенную защиту от ошибок, даже без специального помехоустойчивого кодирования. Именно поэтому фраза понятна даже с опечатками или пропущенными буквами.
